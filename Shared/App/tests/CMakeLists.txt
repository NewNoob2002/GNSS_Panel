cmake_minimum_required(VERSION 3.20)
project(shared_app_core_tests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_subdirectory(.. app)
add_executable(p4_core_test core_tests.cpp)
target_link_libraries(p4_core_test PRIVATE p4_core)
target_compile_options(p4_core_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME p4_core COMMAND p4_core_test)
